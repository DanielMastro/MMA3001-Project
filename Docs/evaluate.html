evaluate.py - measures how accurate and how fast each flux model is.

Where this fits in the project

This file produces the evidence for the report: the model comparison (section 4), the validation (section 5) and the performance analysis (section 6). Nothing in it changes a model; it only measures.

 51def _rms(e):
 52    """Return the root mean square of an array."""
 53    return float(np.sqrt(np.mean(np.square(e))))

 56def _errors(psi_d_hat, psi_q_hat, T_hat, pts, base, torque_col="Torque_Nm"):
 57    """Compute flux and torque errors of predictions against FEA points.
 58
 59    Parameters
 60    ----------
 61    psi_d_hat : numpy.ndarray
 62        Predicted d-axis flux [Wb].
 63    psi_q_hat : numpy.ndarray
 64        Predicted q-axis flux [Wb].
 65    T_hat : numpy.ndarray
 66        Predicted torque [Nm].
 67    pts : pandas.DataFrame
 68        The FEA points in the same order, with Psi_d, Psi_q and torque_col.
 69    base : dict
 70        Per-unit bases from preprocess.base_values().
 71    torque_col : str, optional
 72        Reference torque column. Default "Torque_Nm".
 73
 74    Returns
 75    -------
 76    errors : dict of float
 77        "flux_rmse_pct" and "flux_max_pct" (% of psi_pm), and
 78        "torque_rmse_pct" and "torque_max_pct" (% of the base torque).
 79    """
 80    e_flux = np.r_[psi_d_hat - pts["Psi_d"].to_numpy(),
 81                   psi_q_hat - pts["Psi_q"].to_numpy()] / base["psi_base"] * 100
 82    e_T = (T_hat - pts[torque_col].to_numpy()) / base["T_base"] * 100
 83    return {"flux_rmse_pct": _rms(e_flux), "flux_max_pct": float(np.max(np.abs(e_flux))),
 84            "torque_rmse_pct": _rms(e_T), "torque_max_pct": float(np.max(np.abs(e_T)))}

 87def _fit(make_model, pts):
 88    """Fit a fresh model to the points in pts.
 89
 90    Parameters
 91    ----------
 92    make_model : callable
 93        Returns a new unfitted model, e.g. ``lambda: polynomial_regression(3)``.
 94    pts : pandas.DataFrame
 95        Training points with Id, Iq, Psi_d and Psi_q.
 96
 97    Returns
 98    -------
 99    model : flux model
100        The fitted model.
101    """
102    return make_model().fit(pts["Id"], pts["Iq"], pts["Psi_d"], pts["Psi_q"])

105def _predict(model, pts, pole_pairs):
106    """Predict flux and torque at the operating points in pts.
107
108    Parameters
109    ----------
110    model : flux model
111        A fitted model.
112    pts : pandas.DataFrame
113        Operating points with Id and Iq.
114    pole_pairs : int
115        Number of pole pairs p.
116
117    Returns
118    -------
119    psi_d : numpy.ndarray
120        Predicted d-axis flux [Wb].
121    psi_q : numpy.ndarray
122        Predicted q-axis flux [Wb].
123    torque : numpy.ndarray
124        Predicted torque [Nm].
125    """
126    Id, Iq = pts["Id"].to_numpy(), pts["Iq"].to_numpy()
127    psi_d, psi_q = model.predict_flux(Id, Iq)
128    return psi_d, psi_q, torque(model, Id, Iq, pole_pairs)

134def kfold_cv(make_model, table, pole_pairs, k=5, repeats=3, seed=0):
135    """Score a model by k-fold cross-validation on the unique FEA points.
136
137    The points are shuffled and split into k folds. Each fold is predicted by
138    a model trained on the other k - 1, so every point is scored by a model
139    that never saw it. This is repeated with `repeats` different shuffles and
140    the errors averaged, so one lucky split cannot decide the result.
141
142    Parameters
143    ----------
144    make_model : callable
145        Returns a new unfitted model, e.g. ``lambda: polynomial_regression(6)``.
146    table : pandas.DataFrame
147        Flux map from preprocess.build_flux_map().
148    pole_pairs : int
149        Number of pole pairs p, for the torque error.
150    k : int, optional
151        Number of folds. Default 5.
152    repeats : int, optional
153        Number of shuffles. Default 3.
154    seed : int, optional
155        Seed of the first shuffle; shuffle r uses seed + r. Default 0.
156
157    Returns
158    -------
159    errors : dict of float
160        Means over the shuffles of "flux_rmse_pct" and "flux_max_pct" (% of
161        psi_pm), and "torque_rmse_pct" and "torque_max_pct" (% of the base
162        torque).
163    """
164    pts = pp.unique_operating_points(table)
165    base = pp.base_values(table, pole_pairs)
166    n = len(pts)
167    results = []
168    for r in range(repeats):
169        order = np.random.default_rng(seed + r).permutation(n)
170        psi_d_hat, psi_q_hat, T_hat = np.empty(n), np.empty(n), np.empty(n)
171        for fold in np.array_split(order, k):
172            train = pts.drop(index=fold)                     # everything except this fold
173            model = _fit(make_model, train)
174            psi_d_hat[fold], psi_q_hat[fold], T_hat[fold] = _predict(model, pts.iloc[fold], pole_pairs)
175        results.append(_errors(psi_d_hat, psi_q_hat, T_hat, pts, base))
176    return {key: float(np.mean([r[key] for r in results])) for key in results[0]}

179def seed_sensitivity(make_model, cobalt, p_cob, zenodo, p_zen, seeds=range(10),
180                     k=5, repeats=3, n_rings=2):
181    """Repeat the main tests of a stochastic model at several random seeds.
182
183    A neural network starts training from random weights, so its accuracy
184    depends on the seed as well as on its size. A deterministic model
185    (constant inductance, lookup table, polynomial) gives the same numbers
186    for every seed, so only the network needs this check.
187
188    Parameters
189    ----------
190    make_model : callable
191        make_model(seed) returns an unfitted model.
192    cobalt : pandas.DataFrame
193        Cobalt flux map.
194    p_cob : int
195        Cobalt pole pairs.
196    zenodo : pandas.DataFrame
197        Zenodo flux map.
198    p_zen : int
199        Zenodo pole pairs.
200    seeds : iterable of int, optional
201        Seeds to try. Default range(10).
202    k : int, optional
203        Cross-validation folds. Default 5.
204    repeats : int, optional
205        Cross-validation shuffles. Default 3.
206    n_rings : int, optional
207        Outer current rings held out. Default 2.
208
209    Returns
210    -------
211    results : pandas.DataFrame
212        One row per seed, with errors in %: seed, cobalt_cv_flux_rmse_pct,
213        cobalt_cv_torque_rmse_pct, cobalt_rings_flux_rmse_pct,
214        zenodo_inside_flux_rmse_pct, zenodo_inside_torque_rmse_pct and
215        zenodo_outside_flux_rmse_pct.
216    """
217    rows = []
218    for s in seeds:
219        make = lambda s=s: make_model(s)                     # bind this seed
220        cv = kfold_cv(make, cobalt, p_cob, k, repeats)
221        rings = held_out_rings(make, cobalt, p_cob, n_rings)
222        z = zenodo_validation({"model": make}, zenodo, p_zen).iloc[0]
223        rows.append({"seed": s,
224                     "cobalt_cv_flux_rmse_pct": cv["flux_rmse_pct"],
225                     "cobalt_cv_torque_rmse_pct": cv["torque_rmse_pct"],
226                     "cobalt_rings_flux_rmse_pct": rings["flux_rmse_pct"],
227                     "zenodo_inside_flux_rmse_pct": z["inside_flux_rmse_pct"],
228                     "zenodo_inside_torque_rmse_pct": z["inside_torque_rmse_pct"],
229                     "zenodo_outside_flux_rmse_pct": z["outside_flux_rmse_pct"]})
230    return pd.DataFrame(rows)

233def select_settings(table, pole_pairs, degrees=(2, 3, 4, 5, 6), sizes=(8, 16, 32),
234                    k=5, repeats=3, seed=0):
235    """Choose the polynomial degree and network size by cross-validation.
236
237    Every setting is scored with kfold_cv() and the one with the lowest
238    cross-validated flux error wins. Run this on Cobalt only: the Zenodo
239    motor is kept for the final, independent check.
240
241    Parameters
242    ----------
243    table : pandas.DataFrame
244        Flux map (Cobalt).
245    pole_pairs : int
246        Number of pole pairs p.
247    degrees : tuple of int, optional
248        Polynomial degrees to try. Default (2, 3, 4, 5, 6).
249    sizes : tuple of int, optional
250        Network sizes (neurons per hidden layer) to try. Default (8, 16, 32).
251    k : int, optional
252        Cross-validation folds. Default 5.
253    repeats : int, optional
254        Cross-validation shuffles. Default 3.
255    seed : int, optional
256        Seed of the first shuffle. Default 0.
257
258    Returns
259    -------
260    results : pandas.DataFrame
261        One row per setting: model, setting and the kfold_cv() errors.
262    best_degree : int
263        Polynomial degree with the lowest flux error.
264    best_size : int
265        Network size with the lowest flux error.
266    """
267    from models import neural_network, polynomial_regression
268    rows = []
269    for d in degrees:
270        rows.append({"model": "polynomial_regression", "setting": d,
271                     **kfold_cv(lambda d=d: polynomial_regression(d), table, pole_pairs, k, repeats, seed)})
272    for s in sizes:
273        rows.append({"model": "neural_network", "setting": s,
274                     **kfold_cv(lambda s=s: neural_network(s), table, pole_pairs, k, repeats, seed)})
275    res = pd.DataFrame(rows)
276    best = lambda name: int(res.loc[res[res.model == name].flux_rmse_pct.idxmin(), "setting"])
277    return res, best("polynomial_regression"), best("neural_network")

283def held_out_rings(make_model, table, pole_pairs, n_rings=2):
284    """Train without the outer current rings, then test on them.
285
286    The FEA sweep is a set of current-magnitude rings (the J values).
287    Removing the outer rings measures how a model behaves beyond the
288    currents it was trained on (extrapolation). For Cobalt, n_rings = 2
289    holds out the 102 A and 113 A rings.
290
291    Parameters
292    ----------
293    make_model : callable
294        Returns a new unfitted model.
295    table : pandas.DataFrame
296        Flux map from preprocess.build_flux_map().
297    pole_pairs : int
298        Number of pole pairs p.
299    n_rings : int, optional
300        Number of highest-current rings held out. Default 2.
301
302    Returns
303    -------
304    errors : dict of float
305        Errors on the held-out rings, with the same keys as kfold_cv().
306    """
307    pts = pp.unique_operating_points(table)
308    base = pp.base_values(table, pole_pairs)
309    levels = np.sort(pts[pp.COL_J].unique())
310    held = pts[pp.COL_J] >= levels[-n_rings]
311    model = _fit(make_model, pts[~held])
312    test = pts[held]
313    return _errors(*_predict(model, test, pole_pairs), test, base)

316def compare_models(factories, table, pole_pairs, k=5, repeats=3, n_rings=2):
317    """Build the alternatives table: every model, inside and outside the data.
318
319    Each model is cross-validated (interpolation) and tested on the
320    held-out outer rings (extrapolation).
321
322    Parameters
323    ----------
324    factories : dict
325        {name: make_model}, where make_model() returns an unfitted model.
326    table : pandas.DataFrame
327        Flux map from preprocess.build_flux_map().
328    pole_pairs : int
329        Number of pole pairs p.
330    k : int, optional
331        Cross-validation folds. Default 5.
332    repeats : int, optional
333        Cross-validation shuffles. Default 3.
334    n_rings : int, optional
335        Outer rings held out. Default 2.
336
337    Returns
338    -------
339    results : pandas.DataFrame
340        One row per model: the kfold_cv() errors prefixed "cv_" and the
341        held_out_rings() errors prefixed "rings_".
342    """
343    rows = []
344    for name, make in factories.items():
345        cv = kfold_cv(make, table, pole_pairs, k, repeats)
346        ring = held_out_rings(make, table, pole_pairs, n_rings)
347        rows.append({"model": name,
348                     **{f"cv_{key}": v for key, v in cv.items()},
349                     **{f"rings_{key}": v for key, v in ring.items()}})
350    return pd.DataFrame(rows)

356def zenodo_split(table, n_train_rings=11, train_angles=(0, 20, 40, 60, 80, 90),
357                 outside_angles=(-20, -10, 100, 110)):
358    """Split the dense Zenodo map so its training set imitates Cobalt's data.
359
360    * train: about n_train_rings current rings at the train_angles in the
361      first quadrant (Id <= 0, Iq >= 0), plus the no-load point. That is
362      roughly the density of the Cobalt FEA (71 points).
363    * inside: every other first-quadrant point up to the highest training
364      ring (interpolation).
365    * outside: the outside_angles, just beyond the training quadrant, at the
366      same currents (extrapolation).
367
368    Parameters
369    ----------
370    table : pandas.DataFrame
371        Zenodo flux map.
372    n_train_rings : int, optional
373        Approximate number of current rings in the training set. Default 11
374        (every 60 A from 0 to 600 A).
375    train_angles : tuple of float, optional
376        Training current angles [deg]. Default (0, 20, 40, 60, 80, 90).
377    outside_angles : tuple of float, optional
378        Extrapolation test angles [deg]. Default (-20, -10, 100, 110).
379
380    Returns
381    -------
382    pts : pandas.DataFrame
383        The unique operating points.
384    train : numpy.ndarray of bool
385        Mask of the training points.
386    inside : numpy.ndarray of bool
387        Mask of the interpolation test points.
388    outside : numpy.ndarray of bool
389        Mask of the extrapolation test points.
390    """
391    pts = pp.unique_operating_points(table)
392    J = pts[pp.COL_J].to_numpy()
393    T = pts[pp.COL_TIMING].round(6).to_numpy()
394    levels = np.sort(np.unique(J))
395    step = max(1, int(round((len(levels) - 1) / (n_train_rings - 1))))
396    train_levels = levels[::step]
397    J_top = train_levels.max()
398    train = (np.isin(J, train_levels) & np.isin(T, train_angles)) | (J == 0)
399    inside = ~train & (T >= 0) & (T <= 90) & (J <= J_top) & (J > 0)
400    outside = np.isin(T, outside_angles) & (J <= J_top) & (J > 0)
401    return pts, train, inside, outside

404def zenodo_validation(factories, table, pole_pairs, **split_kwargs):
405    """Test each frozen model on the independent Zenodo motor.
406
407    Each model is trained on the coarse subset from zenodo_split() only, then
408    tested on:
409
410    * flux at the dense points inside and outside the training region;
411    * torque against the FEA's own torque (Torque_FEA_Nm), which never went
412      through any of this code. That is the independent check;
413    * incremental inductance against Zenodo's dense-grid reference, the
414      Ld_matrix and Lq_matrix columns. For Zenodo these come from 5-point
415      finite differences on the full 10 A x 10 deg grid
416      (create_zenodo_input_data.py), not from Maxwell.
417
418    Parameters
419    ----------
420    factories : dict
421        {name: make_model}, where make_model() returns an unfitted model.
422    table : pandas.DataFrame
423        Zenodo flux map, with Torque_FEA_Nm.
424    pole_pairs : int
425        Number of pole pairs p.
426    **split_kwargs
427        Passed on to zenodo_split() to change the split.
428
429    Returns
430    -------
431    results : pandas.DataFrame
432        One row per model: n_train, n_inside and n_outside; the flux and
433        torque errors inside and outside the training region (prefixes
434        "inside_" and "outside_", keys as in kfold_cv()); and the incremental
435        inductance errors inside it as % of the mean reference value
436        (inside_Ld_rmse_pct, inside_Ld_median_abs_pct and the same for Lq).
437    """
438    pts, train, inside, outside = zenodo_split(table, **split_kwargs)
439    base = pp.base_values(table, pole_pairs)
440    torque_col = "Torque_FEA_Nm" if "Torque_FEA_Nm" in pts else "Torque_Nm"
441    rows = []
442    for name, make in factories.items():
443        model = _fit(make, pts[train])
444        row = {"model": name, "n_train": int(train.sum()),
445               "n_inside": int(inside.sum()), "n_outside": int(outside.sum())}
446        for label, mask in (("inside", inside), ("outside", outside)):
447            sub = pts[mask]
448            err = _errors(*_predict(model, sub, pole_pairs), sub, base, torque_col)
449            row.update({f"{label}_{key}": v for key, v in err.items()})
450        # Incremental inductance inside the training region, vs the dense reference
451        sub = pts[inside]
452        inc = incremental_inductance(model, sub["Id"].to_numpy(), sub["Iq"].to_numpy())
453        for out_key, ref_col in (("Ld", "Ld_matrix"), ("Lq", "Lq_matrix")):
454            ref = sub[ref_col].to_numpy()
455            rel = (inc[out_key] - ref) / np.mean(np.abs(ref)) * 100
456            row[f"inside_{out_key}_rmse_pct"] = _rms(rel)
457            # The median is not dominated by a few points: the lookup table's
458            # slope jumps between triangles and at its nearest-point fallback
459            row[f"inside_{out_key}_median_abs_pct"] = float(np.median(np.abs(rel)))
460        rows.append(row)
461    return pd.DataFrame(rows)

467def torque_formula_comparison(table, pole_pairs):
468    """Compare the flux torque formula with the original matrix formula.
469
470    With FEA torque available (Zenodo), both formulas are scored against it.
471    Without it (Cobalt), they are compared with each other, using the flux
472    formula as the reference.
473
474    Parameters
475    ----------
476    table : pandas.DataFrame
477        Flux map from preprocess.build_flux_map().
478    pole_pairs : int
479        Number of pole pairs p.
480
481    Returns
482    -------
483    results : pandas.DataFrame
484        One row per formula: formula, reference, rmse_Nm, max_abs_Nm,
485        rmse_pct_of_base and peak_torque_Nm.
486    """
487    pts = pp.unique_operating_points(table)
488    base = pp.base_values(table, pole_pairs)
489    rows = []
490    ref_col = "Torque_FEA_Nm" if "Torque_FEA_Nm" in pts else "Torque_Nm"
491    for col, label in (("Torque_Nm", "flux formula"), ("Torque_matrix_Nm", "preprocess_v3 matrix formula")):
492        e = pts[col] - pts[ref_col]
493        rows.append({"formula": label, "reference": ref_col,
494                     "rmse_Nm": _rms(e), "max_abs_Nm": float(e.abs().max()),
495                     "rmse_pct_of_base": _rms(e) / base["T_base"] * 100,
496                     "peak_torque_Nm": float(pts[col].max())})
497    return pd.DataFrame(rows)

500def matrix_inductance_consistency(table):
501    """Check whether psi = psi_pm_eff + L * i reproduces the FEA flux.
502
503    Tested with Maxwell's matrix inductances and with the apparent ones. The
504    apparent inductances reproduce the flux by definition, so their row is a
505    sanity check of the code.
506
507    Parameters
508    ----------
509    table : pandas.DataFrame
510        Flux map from preprocess.build_flux_map().
511
512    Returns
513    -------
514    results : pandas.DataFrame
515        One row per inductance type: the RMS and largest psi_d and psi_q
516        errors, as % of psi_pm.
517    """
518    pts = pp.unique_operating_points(table)
519    psi_pm = pts["Psi_pm"].iloc[0]
520    rows = []
521    for label, Ld, Lq in (("matrix", "Ld_matrix", "Lq_matrix"), ("apparent", "Ld_app", "Lq_app")):
522        e_d = (pts["Psi_pm_eff"] + pts[Ld] * pts["Id"] - pts["Psi_d"]) / psi_pm * 100
523        e_q = (pts[Lq] * pts["Iq"] - pts["Psi_q"]) / psi_pm * 100
524        rows.append({"inductances": label,
525                     "psi_d_rmse_pct": _rms(e_d), "psi_d_max_pct": float(e_d.abs().max()),
526                     "psi_q_rmse_pct": _rms(e_q), "psi_q_max_pct": float(e_q.abs().max())})
527    return pd.DataFrame(rows)

533def time_preprocessing(path, settings, repeats=3):
534    """Time the original row loop against the vectorised Park transform.
535
536    Times the dq transform of the 3x3 inductance matrix, the part of the
537    original script that was rewritten. Each method runs `repeats` times and
538    the fastest run is kept. Also checks that both give the same numbers.
539
540    Parameters
541    ----------
542    path : str or pathlib.Path
543        Winding-table CSV.
544    settings : dict

547        performance function takes the same arguments.

557
558    Notes
559    -----
560    These are wall-clock times, so they change a little from run to run with
561    the computer's load and clock speed.

564    data = pp.rotor_angle(raw, verbose=False)

567        best = np.inf

576    df["speed_up"] = df.loc[df.method == "loop", "seconds"].iloc[0] / df["seconds"]
577    df["max_difference_H"] = float(np.max(np.abs(results["loop"] - results["vectorised"])))

614def model_costs(factories, table, n_queries=100_000, repeats=3, seed=0):
615    """Measure each model's fit time, prediction time, memory and FLOPs.
616
617    Each model is fitted to all points and then predicts n_queries random
618    currents inside the training range. Each timing is the fastest of
619    `repeats` runs.
620
621    Parameters
622    ----------
623    factories : dict
624        {name: make_model}, where make_model() returns an unfitted model.
625    table : pandas.DataFrame
626        Flux map to fit.
627    n_queries : int, optional
628        Number of random prediction points. Default 100 000.
629    repeats : int, optional
630        Timing runs; the fastest is kept. Default 3.
631    seed : int, optional
632        Seed for the random query points. Default 0.
633
634    Returns
635    -------
636    results : pandas.DataFrame
637        One row per model: fit_ms, predict_s_per_million (seconds per million
638        predictions), parameters, memory_kB (8 bytes per parameter) and
639        flops_per_eval (an estimate).
640    """
641    pts = pp.unique_operating_points(table)
642    rng = np.random.default_rng(seed)
643    rows = []
644    for name, make in factories.items():
645        fit_best = np.inf
646        for _ in range(repeats):
647            t0 = time.perf_counter()
648            model = _fit(make, pts)
649            fit_best = min(fit_best, time.perf_counter() - t0)
650        # Random query points inside the FEA range (polar: magnitude x angle)
651        lo, hi = model.beta_range
652        I = model.I_base * np.sqrt(rng.random(n_queries))
653        beta = np.deg2rad(lo + (hi - lo) * rng.random(n_queries))
654        Id, Iq = -I * np.sin(beta), I * np.cos(beta)
655        pred_best = np.inf
656        for _ in range(repeats):
657            t0 = time.perf_counter()
658            model.predict_flux(Id, Iq)
659            pred_best = min(pred_best, time.perf_counter() - t0)
660        rows.append({"model": name, "fit_ms": fit_best * 1e3,
661                     "predict_s_per_million": pred_best * 1e6 / n_queries,
662                     "parameters": model.n_parameters,
663                     "memory_kB": model.n_parameters * 8 / 1024,
664                     "flops_per_eval": model.flops_per_eval()})
665    return pd.DataFrame(rows)

136    """Score a model by k-fold cross-validation on the unique FEA points.
137
138    The points are shuffled and split into k folds. Each fold is predicted by
139    a model trained on the other k - 1, so every point is scored by a model
140    that never saw it. This is repeated with `repeats` different shuffles and
141    the errors averaged, so one lucky split cannot decide the result.
142
143    Parameters
144    ----------
145    make_model : callable
146        Returns a new unfitted model, e.g. ``lambda: polynomial_regression(6)``.
147    table : pandas.DataFrame
148        Flux map from preprocess.build_flux_map().
149    pole_pairs : int
150        Number of pole pairs p, for the torque error.
151    k : int, optional
152        Number of folds. Default 5.
153    repeats : int, optional
154        Number of shuffles. Default 3.
155    seed : int, optional
156        Seed of the first shuffle; shuffle r uses seed + r. Default 0.
157
158    Returns
159    -------
160    errors : dict of float
161        Means over the shuffles of "flux_rmse_pct" and "flux_max_pct" (% of
162        psi_pm), and "torque_rmse_pct" and "torque_max_pct" (% of the base

166    base = pp.base_values(table, pole_pairs)
167    n = len(pts)
168    results = []

170        order = np.random.default_rng(seed + r).permutation(n)

357def zenodo_split(table, n_train_rings=11, train_angles=(0, 20, 40, 60, 80, 90),
358                 outside_angles=(-20, -10, 100, 110)):
359    """Split the dense Zenodo map so its training set imitates Cobalt's data.
360
361    * train: about n_train_rings current rings at the train_angles in the
362      first quadrant (Id <= 0, Iq >= 0), plus the no-load point. That is
363      roughly the density of the Cobalt FEA (71 points).
364    * inside: every other first-quadrant point up to the highest training
365      ring (interpolation).
366    * outside: the outside_angles, just beyond the training quadrant, at the
367      same currents (extrapolation).
368
369    Parameters
370    ----------
371    table : pandas.DataFrame
372        Zenodo flux map.
373    n_train_rings : int, optional
374        Approximate number of current rings in the training set. Default 11
375        (every 60 A from 0 to 600 A).
376    train_angles : tuple of float, optional
377        Training current angles [deg]. Default (0, 20, 40, 60, 80, 90).
378    outside_angles : tuple of float, optional
379        Extrapolation test angles [deg]. Default (-20, -10, 100, 110).
380
381    Returns
382    -------
383    pts : pandas.DataFrame
384        The unique operating points.

386        Mask of the training points.
387    inside : numpy.ndarray of bool
388        Mask of the interpolation test points.
389    outside : numpy.ndarray of bool
390        Mask of the extrapolation test points.
391    """
392    pts = pp.unique_operating_points(table)
393    J = pts[pp.COL_J].to_numpy()
394    T = pts[pp.COL_TIMING].round(6).to_numpy()
395    levels = np.sort(np.unique(J))
396    step = max(1, int(round((len(levels) - 1) / (n_train_rings - 1))))
397    train_levels = levels[::step]
398    J_top = train_levels.max()

400    inside = ~train & (T >= 0) & (T <= 90) & (J <= J_top) & (J > 0)
401    outside = np.isin(T, outside_angles) & (J <= J_top) & (J > 0)
402    return pts, train, inside, outside

407

437        (inside_Ld_rmse_pct, inside_Ld_median_abs_pct and the same for Lq).

451        # Incremental inductance inside the training region, vs the dense reference
452        sub = pts[inside]
453        inc = incremental_inductance(model, sub["Id"].to_numpy(), sub["Iq"].to_numpy())
454        for out_key, ref_col in (("Ld", "Ld_matrix"), ("Lq", "Lq_matrix")):
455            ref = sub[ref_col].to_numpy()

457            row[f"inside_{out_key}_rmse_pct"] = _rms(rel)
458            # The median is not dominated by a few points: the lookup table's
459            # slope jumps between triangles and at its nearest-point fallback
460            row[f"inside_{out_key}_median_abs_pct"] = float(np.median(np.abs(rel)))
461        rows.append(row)
462    return pd.DataFrame(rows)

615def model_costs(factories, table, n_queries=100_000, repeats=3, seed=0):
616    """Measure each model's fit time, prediction time, memory and FLOPs.
617
618    Each model is fitted to all points and then predicts n_queries random
619    currents inside the training range. Each timing is the fastest of
620    `repeats` runs.
621
622    Parameters
623    ----------
624    factories : dict
625        {name: make_model}, where make_model() returns an unfitted model.
626    table : pandas.DataFrame
627        Flux map to fit.
628    n_queries : int, optional
629        Number of random prediction points. Default 100 000.
630    repeats : int, optional
631        Timing runs; the fastest is kept. Default 3.
632    seed : int, optional
633        Seed for the random query points. Default 0.
634
635    Returns
636    -------
637    results : pandas.DataFrame
638        One row per model: fit_ms, predict_s_per_million (seconds per million
639        predictions), parameters, memory_kB (8 bytes per parameter) and
640        flops_per_eval (an estimate).
641    """
642    pts = pp.unique_operating_points(table)
643    rng = np.random.default_rng(seed)
644    rows = []
645    for name, make in factories.items():
646        fit_best = np.inf
647        for _ in range(repeats):
648            t0 = time.perf_counter()
649            model = _fit(make, pts)
650            fit_best = min(fit_best, time.perf_counter() - t0)
651        # Random query points inside the FEA range (polar: magnitude x angle)
652        lo, hi = model.beta_range
653        I = model.I_base * np.sqrt(rng.random(n_queries))
654        beta = np.deg2rad(lo + (hi - lo) * rng.random(n_queries))
655        Id, Iq = -I * np.sin(beta), I * np.cos(beta)
656        pred_best = np.inf
657        for _ in range(repeats):
658            t0 = time.perf_counter()
659            model.predict_flux(Id, Iq)
660            pred_best = min(pred_best, time.perf_counter() - t0)
661        rows.append({"model": name, "fit_ms": fit_best * 1e3,
662                     "predict_s_per_million": pred_best * 1e6 / n_queries,
663                     "parameters": model.n_parameters,
664                     "memory_kB": model.n_parameters * 8 / 1024,
665                     "flops_per_eval": model.flops_per_eval()})
666    return pd.DataFrame(rows)
